# EPIC-11 STORY-11.1 — Admin UI app shell, auth, tenant switcher — Implementation Plan

> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** Stand up the reference Admin UI shell — a React/TS SPA served same-origin by `ragctl` at `/admin` — with session login/logout, session hydration, and a tenant switcher.

**Architecture:** Vite + React + TypeScript SPA in `web/`, built to `web/dist/` and `go:embed`-ed into `ragctl`, served at `GET /admin/{path...}` with `index.html` fallback. Auth reuses the existing session cookie (`rag_session`, HttpOnly) + CSRF double-submit; a new `GET /v1/auth/me` hydrates the SPA. Later EPIC-11 stories add their screens + endpoints.

**Tech Stack:** Go 1.22+ (`net/http` ServeMux, `go:embed`), pgx; React 18 + TypeScript + Vite, React Router, TanStack Query, CSS Modules; Vitest + React Testing Library; Playwright; mise tasks.

**Spec:** `docs/specs/SPEC-11-admin-ui.md` (+ `docs/adr/0073-admin-ui-spa-architecture.md`)

## Global Constraints

- Tenant content reached ONLY via resolver → `*tenant.DB` (ADR-0003). STORY-11.1 touches no tenant content; it uses the control-plane pool for auth/registry only.
- Session token never leaves the HttpOnly cookie; never log a session token or CSRF secret (SPEC-09 §3).
- `GET /v1/auth/me` is session-authenticated (behind `RequireSession`) but NOT platform-admin gated — any logged-in user may call it. It is a GET, so no CSRF.
- Same-origin: no CORS added; the SPA calls `/v1/*` and `/admin/tenants` on the same origin.
- `go:embed web/dist` must always compile: a placeholder `web/dist/index.html` is committed; the `build` mise task builds the web app before `go build`.
- Node pinned via mise; the Go build must not require Node at `go build` time (the committed/real `dist` is what embeds).
- Every change carries an issue: **ISSUE-0056** (created in Task 1). Commit messages: no co-authored/signed-off trailers.
- New/changed Go files must pass `mise run lint` with zero new issues (10 known pre-existing elsewhere stay).

```go
// internal/adminui/adminui.go
package adminui

import ("embed"; "io/fs"; "net/http"; "strings")

//go:embed all:dist
var dist embed.FS

// Handler serves the embedded SPA. Real files (JS/CSS/assets) are served as-is;
// any other path under /admin falls back to index.html so client-side routing
// works (SPEC-11 §1). It is mounted at "GET /admin/{path...}", so more specific
// /admin JSON routes (e.g. GET /admin/tenants) win by ServeMux precedence.
func Handler() http.Handler {
	sub, err := fs.Sub(dist, "dist")
	if err != nil {
		panic(err) // dist is embedded at build time; absence is a build error
	}
	fileServer := http.FileServer(http.FS(sub))
	return http.HandlerFunc(func(w http.ResponseWriter, r *http.Request) {
		// Strip the /admin prefix the mux matched, so paths resolve inside dist.
		p := strings.TrimPrefix(r.URL.Path, "/admin")
		p = strings.TrimPrefix(p, "/")
		if p == "" {
			p = "index.html"
		}
		if _, err := fs.Stat(sub, p); err != nil {
			// Not a real asset → client-side route → serve index.html.
			r2 := r.Clone(r.Context())
			r2.URL.Path = "/"
			fileServer.ServeHTTP(w, r2)
			return
		}
		r2 := r.Clone(r.Context())
		r2.URL.Path = "/" + p
		fileServer.ServeHTTP(w, r2)
	})
}
```

```go
// internal/cp/auth/me.go
package auth

import "context"

type MeUser struct { ID, Email string }
type MeMembership struct { TenantID, Slug, Name string; Role Role }
type MeView struct {
	User MeUser; IsPlatformAdmin bool; Memberships []MeMembership
}

// MeService resolves the current user and their tenant memberships for the admin
// UI (SPEC-11 §2.1). Control-plane registry/auth data only — no tenant content.
type MeService struct { DB MembershipDB }

func NewMeService(db MembershipDB) *MeService { return &MeService{DB: db} }

func (s *MeService) Me(ctx context.Context, userID string) (MeView, error) {
	var v MeView
	v.User.ID = userID
	if err := s.DB.QueryRow(ctx,
		`select email, is_platform_admin from users where id = $1`, userID,
	).Scan(&v.User.Email, &v.IsPlatformAdmin); err != nil {
		return MeView{}, err
	}
	rows, err := s.DB.Query(ctx, `
		select t.id::text, t.slug, t.name, m.role
		from tenant_members m join tenants t on t.id = m.tenant_id
		where m.user_id = $1 order by t.name`, userID)
	if err != nil { return MeView{}, err }
	defer rows.Close()
	for rows.Next() {
		var m MeMembership; var role string
		if err := rows.Scan(&m.TenantID, &m.Slug, &m.Name, &role); err != nil { return MeView{}, err }
		m.Role = Role(role); v.Memberships = append(v.Memberships, m)
	}
	return v, rows.Err()
}
```

```go
// internal/cp/auth/me_handler.go
package auth

import ("encoding/json"; "net/http")

type MeHandlers struct { Service *MeService }

func (h *MeHandlers) Me(w http.ResponseWriter, r *http.Request) {
	sess, ok := SessionFrom(r.Context())
	if !ok { writeError(w, http.StatusUnauthorized, "no session"); return }
	v, err := h.Service.Me(r.Context(), sess.UserID)
	if err != nil { writeError(w, http.StatusInternalServerError, "failed to load session user"); return }
	type membershipJSON struct { TenantID, Slug, Name, Role string }
	out := struct {
		User struct{ ID, Email string } `json:"user"`
		IsPlatformAdmin bool `json:"is_platform_admin"`
		Memberships []membershipJSON `json:"memberships"`
		CSRFToken string `json:"csrf_token"`
	}{ CSRFToken: sess.CSRFToken, IsPlatformAdmin: v.IsPlatformAdmin }
	out.User.ID = v.User.ID; out.User.Email = v.User.Email
	for _, m := range v.Memberships {
		out.Memberships = append(out.Memberships, membershipJSON{m.TenantID, m.Slug, m.Name, string(m.Role)})
	}
	w.Header().Set("Content-Type", "application/json")
	_ = json.NewEncoder(w).Encode(out)
}
```

```ts
// web/src/api/client.ts
export class Unauthorized extends Error {}
type Opts = RequestInit & { csrfToken?: string }
export async function apiFetch(path: string, opts: Opts = {}): Promise<Response> {
  const { csrfToken, headers, ...rest } = opts
  const h = new Headers(headers)
  const mutating = rest.method && !['GET', 'HEAD', 'OPTIONS'].includes(rest.method.toUpperCase())
  if (mutating && csrfToken) h.set('X-CSRF-Token', csrfToken)
  const res = await fetch(path, { ...rest, headers: h, credentials: 'same-origin' })
  if (res.status === 401) throw new Unauthorized()
  return res
}
```
